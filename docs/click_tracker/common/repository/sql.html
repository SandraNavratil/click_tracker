 41    @asynccontextmanager
 42    @override
 43    async def unit_of_work(self) -> AsyncIterator[Self]:
 44        """Provide an async session as a unit of work; nested units are not supported.
 45
 46        Returns:
 47            Self: Unit of work.
 48
 49        Raises:
 50            RuntimeError: When called while already inside a unit of work.
 51        """
 52        if self._current_transaction:
 53            raise RuntimeError("Nested unit of work is not supported")
 54        try:
 55            async with self.connection.get_session() as session:
 56                self._current_transaction = session
 57                yield self
 58        finally:
 59            self._current_transaction = None

 61    @override
 62    async def get_user(self, user_id: UUID) -> User:
 63        """Get user from DB.
 64
 65        Args:
 66            user_id: User id.
 67
 68        Returns:
 69            User: User.
 70
 71        Raises:
 72            MissingRequiredAttribute: When not called within a unit of work.
 73            EntityNotFound: When no user exists for the given user_id.
 74        """
 75        if not self._current_transaction:
 76            raise MissingRequiredAttribute("Session is required to get user")
 77        result = await self._current_transaction.execute(
 78            select(DBUser).where(DBUser.id == user_id)
 79        )
 80        db_user = result.scalar_one_or_none()
 81        if db_user is None:
 82            raise EntityNotFound("User", str(user_id))
 83        return User.model_validate(db_user)

124    async def get_clicks_with_user_by_user_id(
125        self, user_id: UUID

127        """Get all clicks for a user with user data from DB.

130            user_id: User id.

140            raise MissingRequiredAttribute(

145        )
146        db_user = result.scalar_one_or_none()
147        if db_user is None:
148            raise EntityNotFound("User", str(user_id))
149        user = User.model_validate(db_user)

152            select(DBClick)

157        return [

162            for db_click in db_clicks
